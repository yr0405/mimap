        async function addNewPlace() {
            const nameInput = document.getElementById('new-place-name');
            const noteInput = document.getElementById('new-place-note');
            const name = nameInput.value.trim();
            const note = noteInput.value.trim();

            if (!name) {
                showMessage('請輸入景點名稱！', true);
                return;

            btn.disabled = true;

                // 使用 Nominatim 開源地理編碼服務搜尋座標
                const response = await fetch(`https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(name)}&limit=1`);
                const data = await response.json();

                let lat = 26.2123; // 預設沖繩市區座標
                let lng = 127.6792;
                
                if (data && data.length > 0) {
                    lat = parseFloat(data[0].lat);
                    lng = parseFloat(data[0].lon);
                    showMessage(`成功加入「${name}」！`);
                } else {
                    showMessage(`找不到精確座標，已暫時放置於沖繩市區。`, true);
                }

                const newPlace = {
                    id: Date.now(),
                    name: name,
                    lat: lat,
                    lng: lng,
                    note: note,
                    icon: "fa-map-pin",
                    color: "text-teal-600",
                    bg: "bg-teal-100"
                };

                // 將景點加入當前所在的標籤頁

                } else {
                    appState.days[currentTab].push(newPlace);
                }

                saveState();
                renderTabs();
                renderItinerary();
                
                nameInput.value = '';
                noteInput.value = '';

            }

        function moveItemToDay(fromTab, index, toTabStr) {
            const toTab = parseInt(toTabStr);
            if(isNaN(toTab) || fromTab === toTab) return;

            // 取出項目
            let item;
            if (fromTab === -1) {
                item = appState.unscheduled.splice(index, 1)[0];
            } else {
                item = appState.days[fromTab].splice(index, 1)[0];
            }

            // 放入目標
            if (toTab === -1) {
                appState.unscheduled.push(item);
            } else {
                appState.days[toTab].push(item);
            }

            saveState();
            renderTabs();     
            renderItinerary(); 
            showMessage("景點移動成功！");
        }

        function reorderItem(tab, index, direction) {
            let list = tab === -1 ? appState.unscheduled : appState.days[tab];
            if (direction === -1 && index > 0) {
                // 向上
                [list[index - 1], list[index]] = [list[index], list[index - 1]];
            } else if (direction === 1 && index < list.length - 1) {
                // 向下
                [list[index + 1], list[index]] = [list[index], list[index + 1]];
            }
            saveState();
            renderItinerary();
        }
        
        function deleteItem(tab, index) {
            showConfirm("確定要刪除這個景點嗎？", () => {
                if (tab === -1) {

                } else {
                    appState.days[tab].splice(index, 1);
                }
                saveState();
                renderTabs();
                renderItinerary();
                showMessage("景點已刪除");
            });
        }

        function updateMap(list) {
            clearMap();
            const bounds = [];
            const latlngs = [];

            list.forEach((place, index) => {
                if (place.lat && place.lng) {
                    const latlng = [place.lat, place.lng];
                    latlngs.push(latlng);
                    bounds.push(latlng);

                    // 待安排使用點標記，有排程的加上數字
                    const markerText = currentTab === -1 ? '<i class="fa-solid fa-location-dot"></i>' : (index + 1);
                    const bgColor = currentTab === -1 ? 'bg-amber-500' : 'bg-blue-600';
                    const markerHtml = `<div class="${bgColor} text-white w-6 h-6 rounded-full flex justify-center items-center font-bold shadow-lg border-2 border-white text-xs">${markerText}</div>`;
                    
                    const customIcon = L.divIcon({ html: markerHtml, className: '', iconSize: [24, 24], iconAnchor: [12, 12] });
                    const marker = L.marker(latlng, {icon: customIcon})
                        .bindPopup(`<b class="text-sm">${place.name}</b>`)
                        .addTo(map);
                    
                    currentMarkers.push(marker);
                }
            });

            if (currentTab !== -1 && latlngs.length > 1) {
                currentPolyline = L.polyline(latlngs, { color: '#3b82f6', weight: 4, opacity: 0.8, dashArray: '8, 8', lineJoin: 'round' }).addTo(map);
            }

            setTimeout(() => {
                map.invalidateSize();
                if (bounds.length > 0) {
                    map.fitBounds(L.latLngBounds(bounds), { padding: [30, 30], maxZoom: 15 });
                }
            }, 100);
        }
